
      return res;
    } catch (e) {
      // AbortError = l'utilisateur a annulÃ©, ne jamais retenter dans ce cas.
      if (e.name === "AbortError") throw e;

      lastError = e;
      if (attempt < maxRetries) {
        console.warn(`Tentative ${attempt + 1}/${maxRetries + 1} Ã©chouÃ©e (${e.message}), nouvelle tentative dans ${delayMs}ms...`);
        await new Promise(r => setTimeout(r, delayMs));
        continue;
      }
    }
  }

  throw lastError || new Error("network");
}
